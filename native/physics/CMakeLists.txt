cmake_minimum_required(VERSION 3.24)
project(cordel_physics VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(FetchContent)
set(CORDEL_PHYSICS_BACKENDS "Jolt;Bullet" CACHE STRING "Candidate backends; host selects exactly one")
if("Jolt" IN_LIST CORDEL_PHYSICS_BACKENDS)
  foreach(flag OVERRIDE_CXX_FLAGS INTERPROCEDURAL_OPTIMIZATION GENERATE_DEBUG_SYMBOLS ENABLE_ALL_WARNINGS ENABLE_INSTALL ENABLE_OBJECT_STREAM DEBUG_RENDERER_IN_DEBUG_AND_RELEASE PROFILER_IN_DEBUG_AND_RELEASE TARGET_UNIT_TESTS TARGET_HELLO_WORLD TARGET_PERFORMANCE_TEST TARGET_SAMPLES TARGET_VIEWER USE_SSE4_1 USE_SSE4_2 USE_AVX USE_AVX2 USE_AVX512 USE_LZCNT USE_TZCNT USE_F16C USE_FMADD)
    set(${flag} OFF CACHE BOOL "Minimal portable physics comparison" FORCE)
  endforeach()
  set(CPP_EXCEPTIONS_ENABLED ON CACHE BOOL "Host exceptions remain enabled" FORCE)
  set(CPP_RTTI_ENABLED ON CACHE BOOL "Host RTTI remains enabled" FORCE)
  set(FLOATING_POINT_EXCEPTIONS_ENABLED OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(jolt
    URL https://codeload.github.com/jrouwe/JoltPhysics/tar.gz/0373ec0dd762e4bc2f6acdb08371ee84fa23c6db
    URL_HASH SHA256=834644d7e9dffee9be0220748015667649d3361611b01c7e6839a5eb58e89711
    SOURCE_SUBDIR Build DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(jolt)
  add_library(cordel_physics_jolt src/common/world.cpp src/common/fixture.cpp src/jolt/backend.cpp)
  target_link_libraries(cordel_physics_jolt PRIVATE Jolt)
endif()
if("Bullet" IN_LIST CORDEL_PHYSICS_BACKENDS)
  foreach(flag BUILD_BULLET3 BUILD_BULLET2_DEMOS BUILD_CPU_DEMOS BUILD_OPENGL3_DEMOS BUILD_EXTRAS BUILD_UNIT_TESTS BUILD_PYBULLET BUILD_ENET BUILD_CLSOCKET INSTALL_LIBS INSTALL_CMAKE_FILES USE_GLUT USE_GRAPHICAL_BENCHMARK)
    set(${flag} OFF CACHE BOOL "Minimal physics comparison" FORCE)
  endforeach()
  FetchContent_Declare(bullet
    URL https://codeload.github.com/bulletphysics/bullet3/tar.gz/2c204c49e56ed15ec5fcfa71d199ab6d6570b3f5
    URL_HASH SHA256=f7c486f10b9f645d0577249d02855e076e8368babaa3d31ced6095e7162baee9
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(bullet)
  add_library(cordel_physics_bullet src/common/world.cpp src/common/fixture.cpp src/bullet/backend.cpp)
  target_include_directories(cordel_physics_bullet SYSTEM PRIVATE ${bullet_SOURCE_DIR}/src)
  target_link_libraries(cordel_physics_bullet PRIVATE BulletDynamics BulletCollision LinearMath)
endif()
foreach(backend IN LISTS CORDEL_PHYSICS_BACKENDS)
  string(TOLOWER ${backend} lower)
  target_include_directories(cordel_physics_${lower} PUBLIC include PRIVATE src/common)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cordel_physics_${lower} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
include(CTest)
option(CORDEL_PHYSICS_TESTS "Build shared backend tests and benchmarks" ON)
if(BUILD_TESTING AND CORDEL_PHYSICS_TESTS)
  foreach(backend IN LISTS CORDEL_PHYSICS_BACKENDS)
    string(TOLOWER ${backend} lower)
    add_executable(cordel-physics-${lower} tests/test_queries.cpp)
    target_link_libraries(cordel-physics-${lower} PRIVATE cordel_physics_${lower})
    target_include_directories(cordel-physics-${lower} PRIVATE ../phase1_host/include)
    add_test(NAME physics_${lower} COMMAND cordel-physics-${lower})
  endforeach()
endif()
